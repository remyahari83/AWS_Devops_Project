Steps to create the open ai key as a Kubernetes secret and access it in the container for qa-evaluator application. 

Create a secret.
kubectl create secret generic qa-evaluator-secrets \
  -n qa \
  --from-literal=OPENAI_API_KEY='Key'

verify

kubectl get secret qa-evaluator-secrets -n qa


Add the secret in container:


container {
  name  = "qa-evaluator"
  image = "521765619271.dkr.ecr.us-east-1.amazonaws.com/myapp:latest"

  image_pull_policy = "Always"

  port {
    container_port = 8000
  }

  env {
    name = "OPENAI_API_KEY"

    value_from {
      secret_key_ref {
        name = "qa-evaluator-secrets"
        key  = "OPENAI_API_KEY"
      }
    }
  }

  resources {
    requests = {
      cpu    = "250m"
      memory = "512Mi"
    }

    limits = {
      cpu    = "500m"
      memory = "1Gi"
    }
  }
}
